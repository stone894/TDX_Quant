def safe_float(val, default=0):
    """安全转换为 float，兼容 list（取第一个元素）和 None"""
    if val is None:
        return default
    if isinstance(val, list):
        return float(val[0]) if len(val) > 0 else default
    try:
        return float(val)
    except (ValueError, TypeError):
        return default

def get_limit_price_ratio(code):
    """根据代码前缀返回涨跌停幅度"""
    if code.startswith('68') or code.startswith('30'):
        return 0.20   # 科创板 / 创业板 20%
    elif code.startswith('8'):
        return 0.30   # 北交所 30%
    else:
        return 0.10   # 主板 10%

def is_a_share_stock(code):
    """判断是否为A股股票（排除ETF、可转债、基金、B股等）"""
    pure = code.split('.')[0]
    if len(pure) != 6:
        return False
    prefix3 = pure[:3]
    # 沪市主板 600/601/603/605
    if prefix3 in ('600', '601', '603', '605'):
        return True
    # 深市主板 000/001/002/003
    if prefix3 in ('000', '001', '002', '003'):
        return True
    # 创业板 300/301
    if prefix3 in ('300', '301'):
        return True
    # 科创板 688/689
    if prefix3 in ('688', '689'):
        return True
    # 北交所 430/830/870
    if prefix3 in ('430', '830', '870'):
        return True
    return False

def save_to_csv(data_list, filename, fieldnames):
    """将列表字典写入 CSV 文件（utf-8-sig 编码，Excel 可直接打开）"""
    with open(filename, 'w', newline='', encoding='utf-8-sig') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames, extrasaction='ignore')
        writer.writeheader()
        writer.writerows(data_list)
    print(f"✅ 已保存 {len(data_list)} 条记录 → {filename}")

def _quiet_call(func, *args, **kwargs):
    """调用 TDX API 并静默内部错误输出"""
    _out, _err = sys.stdout, sys.stderr
    sys.stdout = sys.stderr = _devnull
    try:
        return func(*args, **kwargs)
    except Exception:
        return None
    finally:
        sys.stdout, sys.stderr = _out, _err

def scan_all_stocks():
    """
    遍历全A股，一次扫描同时识别：
    - 涨停板：现价 ≥ 涨停价
    - 跌停板：现价 ≤ 跌停价
    - 炸板：盘中最高价触及涨停价，但收盘未封住
    返回 (limit_up_list, limit_down_list, zha_ban_list)
    """
    print("📊 [1/3] 扫描全A股：涨停 / 跌停 / 炸板...")

    raw_list = tq.get_stock_list('5')
    if not raw_list:
        print("❌ 获取股票列表失败，请检查客户端连接")
        return [], [], []
    all_stocks = [c for c in raw_list if is_a_share_stock(c)]
    print(f"   全A股共 {len(all_stocks)} 只（原始 {len(raw_list)}，已过滤非股票证券）")

    # 探测快照是否可用（测试多只活跃股票避免误判）
    use_snapshot = False
    _test_pool = ['600519.SH', '000001.SZ', '600036.SH', '000858.SZ', '601318.SH']
    _test_codes = [c for c in _test_pool if c in set(all_stocks)]
    if not _test_codes:
        _test_codes = all_stocks[:5]
    for _tc in _test_codes:
        try:
            _ts = tq.get_market_snapshot(stock_code=_tc, field_list=[])
            if _ts and isinstance(_ts, dict):
                _tp = safe_float(_ts.get('Now', _ts.get('Price', _ts.get('price', 0))))
                _tpc = safe_float(_ts.get('LastClose', _ts.get('PreClose', _ts.get('pre_close', 0))))
                if _tp > 0 and _tpc > 0:
                    use_snapshot = True
                    break
        except Exception:
            continue

    print(f"   模式：{'快照' if use_snapshot else 'K线 + 逐只快照修正'}")

    limit_up_list = []
    limit_down_list = []
    zha_ban_list = []
    error_count = 0

    # ---- 涨跌家数统计 ----
    total_counted = 0          # 有效统计的股票数
    up_count = down_count = flat_count = 0
    up_gt5 = up_gt7 = down_gt5 = down_gt7 = 0   # 涨跌幅分档（累计口径）

    for i, code in enumerate(all_stocks):
        try:
            name = ''
            price = 0
            pre_close = 0
            amount = 0
            vol = 0
            high = 0
            low = 0
            sellp1 = -1   # 卖一价，-1 表示未知
            buyp1 = -1    # 买一价，-1 表示未知

            if use_snapshot:
                snap = tq.get_market_snapshot(stock_code=code, field_list=[])
                if snap is None or not isinstance(snap, dict):
                    continue
                name      = snap.get('Name', snap.get('name', ''))
                price     = safe_float(snap.get('Now', snap.get('Price', snap.get('price', 0))))
                pre_close = safe_float(snap.get('LastClose', snap.get('PreClose', snap.get('pre_close', 0))))
                vol       = safe_float(snap.get('Volume', snap.get('Vol', snap.get('vol', 0))))
                amount    = safe_float(snap.get('Amount', snap.get('amount', 0)))
                high      = safe_float(snap.get('Max', snap.get('High', snap.get('high', 0))))
                low       = safe_float(snap.get('Min', snap.get('Low', snap.get('low', 0))))
                sellp1    = safe_float(snap.get('Sellp', snap.get('sellp', 0)))
                buyp1     = safe_float(snap.get('Buyp', snap.get('buyp', 0)))
            else:
                data = tq.get_market_data(
                    field_list=[], stock_list=[code],
                    start_time='', end_time='',
                    count=2, dividend_type='none',
                    period='1d', fill_data=False
                )
                if data is None or not isinstance(data, dict):
                    continue
                if 'Close' not in data:
                    continue

                close_df = data['Close']
                col = code if code in close_df.columns else (
                    close_df.columns[0] if len(close_df.columns) == 1 else None
                )
                if col is None or len(close_df) < 1:
                    continue

                vals = close_df[col].dropna().values
                if len(vals) < 1:
                    continue

                price = float(vals[-1])

                # 优先从 PreClose 字段获取昨收；若不存在则回退到前一根K线的Close
                pre_close = 0
                if 'PreClose' in data:
                    pc_df = data['PreClose']
                    pcol = code if code in pc_df.columns else (
                        pc_df.columns[0] if len(pc_df.columns) == 1 else None)
                    if pcol and len(pc_df) > 0:
                        pre_close = float(pc_df[pcol].dropna().values[-1])
                if pre_close <= 0 and len(vals) >= 2:
                    pre_close = float(vals[-2])

                                # K线的昨收/现价可能取到错误日期，用快照修正
                # （快照字段：Now=现价, LastClose=昨收, Max=最高, Min=最低）
                try:
                    _snap = tq.get_market_snapshot(stock_code=code, field_list=[])
                except Exception:
                    _snap = None
                if _snap and isinstance(_snap, dict):
                    _n = _snap.get('Name', _snap.get('name', ''))
                    if _n:
                        name = _n
                    _p  = safe_float(_snap.get('Now', _snap.get('Price', _snap.get('price', 0))))
                    _pc = safe_float(_snap.get('LastClose', _snap.get('PreClose', _snap.get('pre_close', 0))))
                    _hi = safe_float(_snap.get('Max', _snap.get('High', _snap.get('high', 0))))
                    _lo = safe_float(_snap.get('Min', _snap.get('Low', _snap.get('low', 0))))
                    if _p > 0:
                        price = _p
                    if _pc > 0:
                        pre_close = _pc
                    if _hi > 0:
                        high = _hi
                    if _lo > 0:
                        low = _lo

                if pre_close <= 0:
                    continue

                # 最高价/最低价：仅在快照未提供时从K线取
                if high <= 0 and 'High' in data and col in data['High'].columns:
                    h = data['High'][col].dropna().values
                    high = float(h[-1]) if len(h) > 0 else 0
                if low <= 0 and 'Low' in data and col in data['Low'].columns:
                    lo = data['Low'][col].dropna().values
                    low = float(lo[-1]) if len(lo) > 0 else 0

                if 'Volume' in data and col in data['Volume'].columns:
                    v = data['Volume'][col].dropna().values
                    vol = float(v[-1]) if len(v) > 0 else 0
                if 'Amount' in data and col in data['Amount'].columns:
                    a = data['Amount'][col].dropna().values
                    amount = float(a[-1]) if len(a) > 0 else 0

            if pre_close <= 0 or price <= 0:
                continue

            pct_chg = round((price - pre_close) / pre_close * 100, 2)

            # ---- 涨跌家数统计（全A股，含ST，未剔除涨跌停）----
            total_counted += 1
            if pct_chg > 0:
                up_count += 1
                if pct_chg > 5: up_gt5 += 1
                if pct_chg > 7: up_gt7 += 1
            elif pct_chg < 0:
                down_count += 1
                if pct_chg < -5: down_gt5 += 1
                if pct_chg < -7: down_gt7 += 1
            else:
                flat_count += 1

            ratio = get_limit_price_ratio(code)
            limit_up_price   = round(pre_close * (1 + ratio), 2)
            limit_down_price = round(pre_close * (1 - ratio), 2)

            # 获取名称（K线模式无名称时）
            def _ensure_name():
                nonlocal name
                if not name:
                    try:
                        info = tq.get_stock_info(stock_code=code, field_list=[])
                        if info and isinstance(info, dict):
                            name = info.get('Name', info.get('name', code))
                    except Exception:
                        name = code

            base_row = {
                '代码': code,
                '名称': name,
                '现价': price,
                '昨收': pre_close,
                '涨幅%': pct_chg,
                '成交额(万)': round(amount / 10000, 2),
                '成交量(手)': int(vol),
                '涨停类型': f"{int(ratio*100)}cm",
            }

            # ---- 涨停判断 ----
            if price >= limit_up_price:
                _ensure_name()
                if 'ST' in name.upper():
                    continue
                if sellp1 >= 0:
                    is_sealed = (sellp1 == 0 or sellp1 >= limit_up_price)
                    seal_status = '封板' if is_sealed else '开板'
                else:
                    seal_status = '盘后'
                row = {**base_row, '名称': name,
                       '涨停价': limit_up_price, '封板状态': seal_status}
                limit_up_list.append(row)

            # ---- 炸板判断：盘中最高价触及涨停但收盘未封住 ----
            elif high >= limit_up_price and price < limit_up_price:
                _ensure_name()
                if 'ST' in name.upper():
                    continue
                row = {**base_row, '名称': name,
                       '涨停价': limit_up_price, '最高价': high,
                       '回落幅度%': round((high - price) / high * 100, 2)}
                zha_ban_list.append(row)

            # ---- 跌停判断 ----
            if price <= limit_down_price:
                _ensure_name()
                if 'ST' in name.upper():
                    continue
                if buyp1 >= 0:
                    is_sealed = (buyp1 == 0 or buyp1 <= limit_down_price)
                    seal_status = '封板' if is_sealed else '开板'
                else:
                    seal_status = '盘后'
                row = {**base_row, '名称': name,
                       '跌停价': limit_down_price, '封板状态': seal_status}
                limit_down_list.append(row)

        except Exception as e:
            error_count += 1
            if error_count <= 3:
                print(f"   ⚠️ {code} 获取失败: {e}")
            continue

        if (i + 1) % 500 == 0:
            print(f"   已扫描 {i+1}/{len(all_stocks)}，"
                  f"涨停 {len(limit_up_list)} / 跌停 {len(limit_down_list)} / 炸板 {len(zha_ban_list)}")

    # 按成交额降序
    limit_up_list.sort(key=lambda x: x['成交额(万)'], reverse=True)
    limit_down_list.sort(key=lambda x: x['成交额(万)'], reverse=True)
    zha_ban_list.sort(key=lambda x: x['成交额(万)'], reverse=True)

    print(f"   扫描完成！涨停 {len(limit_up_list)} / 跌停 {len(limit_down_list)} / 炸板 {len(zha_ban_list)}")
    print(f"   涨跌家数：上涨 {up_count} / 下跌 {down_count} / 平盘 {flat_count}"
          f"（统计 {total_counted} 只）")
    if error_count > 0:
        print(f"   （{error_count} 只获取失败，已跳过）")

    # 汇总市场情绪指标（单行字典）
    market_stats = {
        '统计股票数': total_counted,
        '上涨家数': up_count,
        '下跌家数': down_count,
        '平盘家数': flat_count,
        '涨跌比': round(up_count / down_count, 2) if down_count > 0 else 0,
        '涨停家数': len(limit_up_list),
        '跌停家数': len(limit_down_list),
        '炸板家数': len(zha_ban_list),
        '涨超5%家数': up_gt5,
        '涨超7%家数': up_gt7,
        '跌超5%家数': down_gt5,
        '跌超7%家数': down_gt7,
    }

    return limit_up_list, limit_down_list, zha_ban_list, market_stats

def get_index_overview():
    """获取主要大盘指数的行情与成交额"""
    print("📊 [2/3] 获取大盘指数行情...")

    index_data = []

    for code, display_name in INDEX_LIST:
        price = pre_close = amount = high = low = open_price = 0

        # 尝试快照（指数在不同环境下字段名可能不同：Price/PreClose 或 Now/LastClose）
        snap = _quiet_call(tq.get_market_snapshot, stock_code=code, field_list=[])
        if snap and isinstance(snap, dict):
            price = safe_float(
                snap.get('Price', snap.get('price', snap.get('Now', snap.get('now', 0))))
            )
            pre_close = safe_float(
                snap.get('PreClose', snap.get('pre_close', snap.get('LastClose', snap.get('last_close', 0))))
            )
            amount = safe_float(snap.get('Amount', snap.get('amount', 0)))
            high = safe_float(
                snap.get('High', snap.get('high', snap.get('Max', snap.get('max', 0))))
            )
            low = safe_float(
                snap.get('Low', snap.get('low', snap.get('Min', snap.get('min', 0))))
            )
            open_price = safe_float(snap.get('Open', snap.get('open', 0)))

        # 快照无效时回退到 K 线
        if price <= 0 or pre_close <= 0:
            data = _quiet_call(tq.get_market_data,
                field_list=[], stock_list=[code],
                start_time='', end_time='',
                count=2, dividend_type='none',
                period='1d', fill_data=False)
            if data and isinstance(data, dict) and 'Close' in data:
                close_df = data['Close']
                col = code if code in close_df.columns else (
                    close_df.columns[0] if len(close_df.columns) == 1 else None)
                if col and len(close_df) >= 2:
                    vals = close_df[col].dropna().values
                    if len(vals) >= 2:
                        pre_close = float(vals[-2])
                        price = float(vals[-1])
                for field, target in [('High', 'high'), ('Low', 'low'),
                                      ('Open', 'open'), ('Amount', 'amount')]:
                    if field in data:
                        df = data[field]
                        c = code if code in df.columns else (
                            df.columns[0] if len(df.columns) == 1 else None)
                        if c and len(df) > 0:
                            v = float(df[c].dropna().values[-1])
                            if field == 'High':    high = v
                            elif field == 'Low':   low = v
                            elif field == 'Open':  open_price = v
                            elif field == 'Amount': amount = v

        pct_chg = round((price - pre_close) / pre_close * 100, 2) if pre_close > 0 else 0

        index_data.append({
            '指数代码': code,
            '指数名称': display_name,
            '收盘': round(price, 2),
            '开盘': round(open_price, 2),
            '最高': round(high, 2),
            '最低': round(low, 2),
            '昨收': round(pre_close, 2),
            '涨跌幅%': pct_chg,
            '涨跌点': round(price - pre_close, 2),
            '成交额(亿)': round(amount / 1e4, 2),
        })
        print(f"   {display_name}: {price:.2f} ({pct_chg:+.2f}%) 成交额 {round(amount/1e4,2)} 亿")

    print(f"   指数行情获取完成，共 {len(index_data)} 个")
    return index_data

def _fetch_category_codes(category):
    """通过 get_stock_list 获取某一类板块/指数的代码与名称
    返回 [{'code':.., 'name':..}, ...]
    """
    raw = _quiet_call(tq.get_stock_list, str(category), list_type=1)
    if not raw:
        return []
    result = []
    for item in raw:
        if isinstance(item, dict):
            code = item.get('Code', item.get('code', ''))
            name = item.get('Name', item.get('name', ''))
        else:
            code, name = str(item), ''
        if code:
            result.append({'code': code, 'name': name})
    return result

def _fetch_quote(code):
    """取单个板块/指数行情：现价、昨收、最高、最低、开盘、成交额"""
    price = pre_close = high = low = open_price = amount = 0.0

    # 优先快照（指数字段多为 Price/PreClose，板块可能为 Now/LastClose）
    snap = _quiet_call(tq.get_market_snapshot, stock_code=code, field_list=[])
    if snap and isinstance(snap, dict):
        price      = safe_float(snap.get('Price', snap.get('Now', snap.get('now', 0))))
        pre_close  = safe_float(snap.get('PreClose', snap.get('LastClose', snap.get('last_close', 0))))
        high       = safe_float(snap.get('High', snap.get('Max', 0)))
        low        = safe_float(snap.get('Low', snap.get('Min', 0)))
        open_price = safe_float(snap.get('Open', snap.get('open', 0)))
        amount     = safe_float(snap.get('Amount', snap.get('amount', 0)))

    # 快照无效时回退 K 线
    if price <= 0 or pre_close <= 0:
        data = _quiet_call(tq.get_market_data,
            field_list=[], stock_list=[code],
            start_time='', end_time='',
            count=2, dividend_type='none',
            period='1d', fill_data=False)
        if data and isinstance(data, dict) and 'Close' in data:
            close_df = data['Close']
            col = code if code in close_df.columns else (
                close_df.columns[0] if len(close_df.columns) == 1 else None)
            if col is not None:
                vals = close_df[col].dropna().values
                if len(vals) >= 1:
                    price = float(vals[-1])
                if 'PreClose' in data:
                    pc_df = data['PreClose']
                    pcol = code if code in pc_df.columns else (
                        pc_df.columns[0] if len(pc_df.columns) == 1 else None)
                    if pcol and len(pc_df) > 0:
                        pre_close = float(pc_df[pcol].dropna().values[-1])
                if pre_close <= 0 and len(vals) >= 2:
                    pre_close = float(vals[-2])
                for field in ('High', 'Low', 'Open', 'Amount'):
                    if field in data:
                        df = data[field]
                        c = code if code in df.columns else (
                            df.columns[0] if len(df.columns) == 1 else None)
                        if c and len(df) > 0:
                            v = float(df[c].dropna().values[-1])
                            if field == 'High':     high = v
                            elif field == 'Low':    low = v
                            elif field == 'Open':   open_price = v
                            elif field == 'Amount': amount = v

    pct_chg = round((price - pre_close) / pre_close * 100, 2) if pre_close > 0 else 0
    return {
        'price': round(price, 2), 'pre_close': round(pre_close, 2),
        'open': round(open_price, 2), 'high': round(high, 2), 'low': round(low, 2),
        'pct_chg': pct_chg, 'amount': amount,
    }

def get_key_index(category=9, type_label='重点指数'):
    """重点指数行情（list_type=9）"""
    print(f"📊 获取 [{type_label}] (list_type={category})...")
    codes = _fetch_category_codes(category)
    if not codes:
        print(f"   ❌ 获取 {type_label} 列表失败，请检查客户端连接")
        return []
    print(f"   共 {len(codes)} 个 {type_label}")

    result = []
    for it in codes:
        q = _fetch_quote(it['code'])
        result.append({
            '指数代码': it['code'],

## 关键设计说明

### 涨停 / 跌停判断

1. 根据代码前缀确定幅度：`60/00` → 10%，`68/30` → 20%，`8` → 30%
2. 涨停价 = `round(昨收 × (1 + 幅度), 2)`，跌停价 = `round(昨收 × (1 - 幅度), 2)`
3. 现价 ≥ 涨停价 → 涨停；现价 ≤ 跌停价 → 跌停
4. 涨停封板状态：通过卖一价判断（卖一 = 0 或 ≥ 涨停价 → 封板）
5. 跌停封板状态：通过买一价判断（买一 = 0 或 ≤ 跌停价 → 封板）
6. ST/*ST 股票不纳入涨停/跌停/炸板统计（通过名称包含 "ST" 判断）
7. K 线回退模式下：先取 K 线数据，再对每只股票调用 `get_market_snapshot` 修正昨收和现价（快照字段映射：`Now`→现价、`LastClose`→昨收、`Max`→最高、`Min`→最低；K 线 `count=2` 在盘后返回 [T-2, T-1] 而非 [T-1, T]，直接用会导致昨收取到前天数据）
8. 股票列表过滤：仅保留 A 股代码（沪市 60xxxx、科创 68xxxx、深市 00xxxx、创业板 30xxxx、北交所 43/83/87xxxx），排除 ETF、可转债、基金、B 股等

### 炸板判断

- 条件：盘中最高价 ≥ 涨停价 **且** 收盘价 < 涨停价
- 快照模式通过 `High` 字段获取盘中最高价；K 线模式通过日 K 的 `High` 获取
- 输出回落幅度 = `(最高价 - 收盘价) / 最高价 × 100`

### 涨跌家数统计

- 在 `scan_all_stocks()` 主循环中复用已算出的 `pct_chg` 累加，零额外取数开销
- `pct_chg > 0` 计入上涨、`\< 0` 计入下跌、`= 0` 计入平盘
- 分档为**累计口径**：涨超 5% 家数包含涨超 7% 家数（跌幅同理）
- 统计范围为全部成功取到行情的 A 股（含 ST，未剔除涨跌停），与涨停/跌停列表口径不同
- 涨跌比 = 上涨家数 / 下跌家数，>1 表示普涨、<1 表示普跌，是衡量当日赚钱效应的核心指标

### 大盘指数

- 覆盖：上证指数、深证成指、创业板指、科创50、中证1000、沪深300
- 优先使用快照获取实时数据，无效时回退到最近 2 根日 K 线
- **成交额单位**：通达信 `get_market_data` 对指数/板块返回的 Amount 单位为 **万元**（个股为元），代码已按 `÷1e4` 换算为亿
- 科创50 代码 `000688.SH`，如客户端不支持可替换为 `931643.CSI`

### 板块数据获取

- 不再枚举猜测代码段，改用 `tq.get_stock_list(market, list_type=1)` 直接获取板块/指数的准确代码与名称列表（与客户端分类一致，不漏不错分）
- `market` 取值：`9` 重点指数、`12` 概念板块；其他常用：`11` 缺省行业、`13` 风格、`14` 地区、`16/17/18` 研究行业一/二/三级
- 拿到代码后用 `get_market_snapshot` / `get_market_data` 补行情，`get_more_info` 补涨停股数、换手率、总市值等扩展数据

### 注意事项

- **板块资金净流入**：待实现。`get_more_info` 对板块指数返回的 `Zjl`/`Zjl_HB` 均为 0，需要找到获取板块成分股的 API 后汇总个股数据
- **指数代码**：如科创50 无数据，尝试 `print(tq.get_market_snapshot('000688.SH', []))` 确认是否生效

### 注意事项

<aside>
⚠️

**字段名与常见命名不同！** 不要用 `Price`/`PreClose`/`High`/`Low`，必须用 `Now`/`LastClose`/`Max`/`Min`。

所有返回值均为**字符串**，需要 `float()` 转换。`Buyp`/`Buyv`/`Sellp`/`Sellv` 为 **list**（5 个元素）。

</aside>

- 盘后调用有效，返回收盘时的快照数据
- `field_list=[]` 返回全部字段
- 对指数代码同样有效（如 `999999.SH`），但字段名可能不同（指数用 `Price`/`PreClose`）
- 每次调用获取单只股票，不支持批量
